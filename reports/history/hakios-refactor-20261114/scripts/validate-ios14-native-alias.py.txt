#!/usr/bin/env python3
"""Check the emitted native alias probe instructions; never execute the kernel.

Mach-O AArch64 assembly treats ';' as a comment. Check the actual AT/ISB/MRS
sequences, since a successful compilation alone did not catch that regression.
Run after validate-ios14-lithium-instructions.py for the same kernel.
"""
import argparse
import hashlib
import json
from pathlib import Path
import re
import subprocess

ROOT = Path(__file__).resolve().parents[1]


def check_sequence(instructions):
    switch = next(i for i, (_, op) in enumerate(instructions)
                  if re.fullmatch(r"bl\s+_set_mmu_ttb_alternate", op))
    relocate = next(i for i, (_, op) in enumerate(instructions)
                    if re.fullmatch(r"bl\s+_lithium_early_relocate", op))
    reads = [i for i in range(max(0, switch - 100), switch)
             if re.match(r"at\s+s1e1r,", instructions[i][1], re.I)]
    writes = [i for i in range(switch, relocate)
              if re.match(r"at\s+s1e1w,", instructions[i][1], re.I)]
    if len(reads) != 1 or len(writes) != 2:
        raise ValueError("expected one bootstrap table probe and two screen probes")
    for i in reads + writes:
        if (not re.fullmatch(r"isb(?:\s+sy)?", instructions[i + 1][1], re.I)
                or not re.fullmatch(r"mrs\s+x\d+,\s*PAR_EL1", instructions[i + 2][1], re.I)
                or instructions[i + 2][0] != instructions[i][0] + 8):
            raise ValueError(f"AT at {instructions[i][0]:#x} must emit ISB and MRS PAR_EL1")
    sequence = [op for _, op in instructions[writes[0] + 3:writes[1]]]
    tlbi = next(i for i, op in enumerate(sequence) if re.fullmatch(r"tlbi\s+vmalle1is", op))
    expected = (r"dsb\s+ishst", r"tlbi\s+vmalle1is", r"dsb\s+ish", r"isb(?:\s+sy)?")
    if tlbi == 0 or len(sequence) <= tlbi + 2 or not all(
        re.fullmatch(pattern, op) for pattern, op in zip(expected, sequence[tlbi - 1:tlbi + 3])
    ):
        raise ValueError("missing ordered DSB/TLBI/DSB/ISB before final alias probe")
    stores = [i for i in range(writes[1] + 3, relocate)
              if re.match(r"strb\s", instructions[i][1])]
    if len(stores) != 1:
        raise ValueError("expected one pixel test after the final translation probe")
    return dict(table_probe=hex(instructions[reads[0]][0]),
                before_tlbi_probe=hex(instructions[writes[0]][0]),
                after_tlbi_probe=hex(instructions[writes[1]][0]),
                pixel_store=hex(instructions[stores[0]][0]),
                relocate_call=hex(instructions[relocate][0]))


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("kernel", type=Path)
    args = parser.parse_args()
    kernel_hash = hashlib.sha256(args.kernel.read_bytes()).hexdigest()
    scan = json.loads((ROOT / "reports/ios14-xnu-lithium-instructions.json").read_text())
    assert scan['kernel_sha256'] == kernel_hash and scan['instruction_policy_passed']
    symbols = []
    for line in subprocess.check_output(['llvm-nm', '-n', '--defined-only', args.kernel], text=True).splitlines():
        parts = line.split()
        if len(parts) == 3:
            symbols.append((int(parts[0], 16), parts[2]))
    start = next(address for address, name in symbols if name == '_arm_vm_init')
    end = min(address for address, _ in symbols if address > start)
    instructions = []
    logs = []
    for section in scan['sections']:
        log = Path(section['log'])
        assert hashlib.sha256(log.read_bytes()).hexdigest() == section['log_sha256']
        logs.append(dict(path=str(log), sha256=section['log_sha256']))
        with log.open() as stream:
            for line in stream:
                match = re.match(r'^([0-9a-f]+):\s+(.+?)\s*$', line)
                if match and start <= int(match[1], 16) < end:
                    instructions.append((int(match[1], 16), match[2]))
    addresses = check_sequence(instructions)
    # Regression control: the old semicolon spelling emitted only AT. Removing
    # the following instructions must be rejected by the same binary check.
    broken = list(instructions)
    bad = next(i for i, (address, _) in enumerate(broken) if address == int(addresses['table_probe'], 16))
    del broken[bad + 1:bad + 3]
    try:
        check_sequence(broken)
    except ValueError:
        pass
    else:
        raise AssertionError('truncated assembly regression was not detected')
    record = dict(kernel_sha256=kernel_hash, instruction_sequences_passed=True,
                  missing_par_read_rejected=True, addresses=addresses, disassembly=logs,
                  qemu_used=False, hardware_verified=False,
                  scope='Emitted AT/ISB/MRS, ordered TLBI and pixel-store sequence only; no execution proof')
    (ROOT / 'reports/ios14-native-alias-validation.json').write_text(json.dumps(record, indent=2) + '\n')
    print(json.dumps({k: v for k, v in record.items() if k != 'disassembly'}))


if __name__ == '__main__':
    main()

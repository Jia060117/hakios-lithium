 * KMA_NOPAGEWAIT here. ZERO/PERMANENT exist for other allocators, but this

 * successful allocation; retain that mapping for this driver's lifetime.
 */
static int __attribute__((noinline))
ufs_alloc_dma(void)
{
    kern_return_t kr = kmem_alloc_contig(kernel_map, &su.dma, DMA_SIZE,
        PAGE_MASK, 0, 0, KMA_NONE, VM_KERN_MEMORY_IOKIT);
    printf("LITHIUM UFS: kmem_alloc_contig kr=%d bytes=0x%x flags=0\\n", kr, DMA_SIZE);

    /* This API does not zero its returned pages. Clear all descriptors and
     * bounce bytes before any host ownership or cache maintenance. */
    bzero((void *)su.dma, DMA_SIZE);
    ppnum_t first = pmap_find_phys(kernel_pmap, (addr64_t)su.dma);

        if (!pn || ptoa_64(pn) != su.dma_pa+off) {
            printf("LITHIUM UFS: DMA physical continuity failed at 0x%lx\\n", (unsigned long)off);
            return EIO;
        }
    }
    printf("LITHIUM UFS: DMA zeroed and physical continuity verified at 0x%llx\\n",su.dma_pa);
    return 0;
}

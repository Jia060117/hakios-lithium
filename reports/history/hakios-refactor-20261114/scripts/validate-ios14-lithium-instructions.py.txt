#!/usr/bin/env python3
"""Scan every executable Mach-O section for instructions absent on MIX Kryo."""
import argparse
import hashlib
import importlib.util
import json
from pathlib import Path
import re
import shutil
import struct
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[1]
FORBIDDEN = re.compile(r"^(?:cas\w*|swp\w*|(?:ld|st)(?:add|clr|eor|set|smax|smin|umax|umin)\w*|"
                       r"pac\w*|aut\w*|xpac\w*|ret(?:aa|ab)|(?:br|blr)(?:aa|ab)\w*)$")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("kernel", type=Path)
    parser.add_argument("--objdump", default=shutil.which("llvm-objdump-18") or shutil.which("llvm-objdump"))
    args = parser.parse_args()
    if not args.objdump:
        parser.error("LLVM Mach-O objdump is required")
    spec = importlib.util.spec_from_file_location("native_xnu", ROOT / "scripts/inspect-native-xnu.py")
    inspector = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(inspector)
    validation = inspector.inspect(args.kernel)
    data = args.kernel.read_bytes()
    offset, sections, data_regions = 32, [], []
    for _ in range(struct.unpack_from("<I", data, 16)[0]):
        cmd, size = struct.unpack_from("<II", data, offset)
        if cmd == 0x19:
            nsections = struct.unpack_from("<I", data, offset + 64)[0]
            if 72 + nsections * 80 > size:
                raise ValueError("section table exceeds load command")
            for i in range(nsections):
                start = offset + 72 + i * 80
                flags = struct.unpack_from("<I", data, start + 64)[0]
                if flags & (0x80000000 | 0x400):
                    section = data[start:start + 16].split(b"\0", 1)[0].decode()
                    segment = data[start + 16:start + 32].split(b"\0", 1)[0].decode()
                    sections.append((segment, section))
        elif cmd == 0x29:  # LC_DATA_IN_CODE; offsets refer to the Mach-O file.
            table, length = struct.unpack_from("<II", data, offset + 8)
            if length % 8 or table + length > len(data):
                raise ValueError("invalid data-in-code table")
            for j in range(length // 8):
                fileoff, bytes_, kind = struct.unpack_from("<IHH", data, table + j * 8)
                segment = next(s for s in validation["segments"] if s["fileoff"] <= fileoff and
                               fileoff + bytes_ <= s["fileoff"] + s["filesize"])
                start = int(segment["vmaddr"], 16) + fileoff - segment["fileoff"]
                data_regions.append(dict(start=start, end=start + bytes_, kind=kind))
        offset += size
    if not sections:
        raise ValueError("kernel contains no instruction sections")
    # This static kernel link omits LC_DATA_IN_CODE. Recover the annotated
    # SHA256 table only after matching all 256 bytes against both the pinned
    # source and the assembled object, then the linked symbol's bytes.
    if not data_regions:
        source = ROOT / "upstream/research/xnu-7195.141.2/osfmk/arm64/corecrypto/sha256_compress_arm64.s"
        table_text = source.read_text().split("K256:\n", 1)[1].split("AccelerateCrypto_SHA256_compress", 1)[0]
        constants = [int(x, 16) for x in re.findall(r"\.long\s+(0x[0-9a-fA-F]+)", table_text)]
        if len(constants) != 64:
            raise ValueError("SHA256 source table must contain exactly 64 round constants")
        expected = struct.pack("<64I", *constants)
        obj = args.kernel.parent / "osfmk/DEVELOPMENT/sha256_compress_arm64.o"
        object_data = obj.read_bytes()
        off, annotation, text_offset = 32, None, None
        for _ in range(struct.unpack_from("<I", object_data, 16)[0]):
            cmd, length = struct.unpack_from("<II", object_data, off)
            if cmd == 0x29:
                tableoff, bytes_ = struct.unpack_from("<II", object_data, off + 8)
                if bytes_ != 8:
                    raise ValueError("unexpected SHA256 data-in-code annotation length")
                annotation = struct.unpack_from("<IHH", object_data, tableoff)
            elif cmd == 0x19:
                text_offset = struct.unpack_from("<I", object_data, off + 72 + 48)[0]
            off += length
        if annotation != (0, 256, 1) or text_offset is None or object_data[text_offset:text_offset + 256] != expected:
            raise ValueError("assembled SHA256 table metadata/bytes disagree with the source")
        named = subprocess.check_output(["llvm-nm", "--defined-only", args.kernel], text=True)
        locations = [int(line.split()[0], 16) for line in named.splitlines() if line.split()[-1:] == ["K256"]]
        if len(locations) != 1:
            raise ValueError("expected one linked SHA256 round-constant table")
        address = locations[0]
        segment = next(s for s in validation["segments"] if int(s["vmaddr"], 16) <= address and
                       address + 256 <= int(s["vmaddr"], 16) + s["filesize"])
        fileoff = address - int(segment["vmaddr"], 16) + segment["fileoff"]
        if data[fileoff:fileoff + 256] != expected:
            raise ValueError("linked SHA256 round constants differ from the annotated object/source")
        data_regions.append(dict(start=address, end=address + 256, kind=1, source=str(source),
                                 source_sha256=hashlib.sha256(source.read_bytes()).hexdigest(),
                                 annotated_object=str(obj), object_sha256=hashlib.sha256(object_data).hexdigest(),
                                 bytes_verified_against_source_and_object=True))
    out = args.kernel.parent / "instruction-check"
    out.mkdir(exist_ok=True)
    records, unsupported, hints, total = [], [], [], 0
    for i, (segment, section) in enumerate(sections):
        command = [args.objdump, "--macho", "--disassemble", "--no-show-raw-insn", "--mattr=+lse,+pauth",
                   f"--section={segment},{section}", str(args.kernel)]
        log = out / f"{i:02d}.txt"
        with log.open("w") as output:
            subprocess.run(command, stdout=output, stderr=subprocess.STDOUT, check=True)
        count = 0
        with log.open() as output:
            for line in output:
                match = re.match(r"^([0-9a-f]+):\s+([a-z][a-z0-9.]*)\b", line)
                if match:
                    address = int(match[1], 16)
                    if any(r["start"] <= address < r["end"] for r in data_regions):
                        continue
                    count += 1
                    if FORBIDDEN.fullmatch(match[2]):
                        item = dict(section=f"{segment},{section}", address="0x" + match[1], instruction=line.strip())
                        if match[2] == "xpaclri":
                            segment_info = next(s for s in validation["segments"] if int(s["vmaddr"], 16) <= address <
                                                int(s["vmaddr"], 16) + s["filesize"])
                            fileoff = address - int(segment_info["vmaddr"], 16) + segment_info["fileoff"]
                            if struct.unpack_from("<I", data, fileoff)[0] != 0xd50320ff:
                                raise ValueError("XPACLRI did not decode from the expected backward-compatible HINT #7 encoding")
                            hints.append(item)  # HINT #7 is a NOP without FEAT_PAuth.
                        else:
                            unsupported.append(item)
        total += count
        records.append(dict(section=f"{segment},{section}", instructions=count,
                            command=command, log=str(log), log_sha256=hashlib.sha256(log.read_bytes()).hexdigest()))
    record = dict(kernel_sha256=validation["sha256"], instruction_count=total, sections=records,
                  forbidden_instructions=unsupported, backward_compatible_hints=hints, data_regions=data_regions,
                  instruction_policy_passed=total > 0 and not unsupported,
                  policy="Exclude LSE atomic and PAC instructions on measured ARMv8.0 Kryo; not a complete ISA/emulation proof.",
                  hardware_verified=False)
    report = ROOT / "reports/ios14-xnu-lithium-instructions.json"
    report.write_text(json.dumps(record, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(dict(kernel_sha256=record["kernel_sha256"], instruction_count=total,
                          forbidden_instruction_count=len(unsupported), compatible_hint_count=len(hints),
                          data_region_count=len(data_regions), passed=record["instruction_policy_passed"])))
    return 0 if record["instruction_policy_passed"] else 1


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""Generate the native screen/boot checkpoints patch from the built 7195 tree.

Run inside WSL after a successful twelve-patch lithium build. Does not modify
the build tree; build-ios14-xnu.py applies the resulting complete series.
"""
import difflib
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
SRC = Path.home() / 'ios-mix1-build/xnu7195-lithium/src'
PATCH = ROOT / 'ports/lithium/patches/xnu-7195/0013-lithium-native-screen-checkpoints.patch'

def main():
    edits = {}
    def get(path):
        if path not in edits:
            edits[path] = [(SRC / path).read_text(), (SRC / path).read_text()]
        return edits[path][1]
    def replace(path, old, new, count=1):
        text = get(path)
        if text.count(old) != count:
            raise ValueError(f'{path}: anchor count {text.count(old)}: {old[:100]!r}')
        edits[path][1] = text.replace(old, new)
    for name in ('lithium_early.c', 'lithium_early.h'):
        edits['osfmk/arm64/' + name] = ['', (ROOT / 'ports/lithium/native' / name).read_text()]
    for path, anchor in (
        ('osfmk/arm/arm_init.c', '#include <arm/proc_reg.h>'),
        ('osfmk/arm64/arm_vm_init.c', '#include <arm64/lowglobals.h>'),
        ('pexpert/arm/pe_serial.c', '#include <arm64/proc_reg.h>')):
        replace(path, anchor, anchor + '\n#include <arm64/lithium_early.h>')
    path = 'osfmk/arm/arm_init.c'
    replace(path, '\tarm_slide_rebase_and_sign_image();',
            '\tlithium_early_mark(1, "arm_init entered; MMU enabled");\n\tarm_slide_rebase_and_sign_image();')
    for anchor, stage, label in (
        ('\tPE_init_platform(FALSE, args); /* Get platform expert set up */', 2, 'platform expert initialized'),
        ('\tml_parse_cpu_topology();', 3, 'CPU topology parsed'),
        ('\tcpu_bootstrap();', 4, 'CPU bootstrap complete'),
        ('\tkernel_startup_bootstrap();', 5, 'kernel startup bootstrap complete'),
        ('\tarm_vm_init(xmaxmem, args);', 15, 'arm_vm_init returned'),
        ('\tPE_init_platform(TRUE, &BootCpuData);', 19, 'interrupt platform initialized')):
        replace(path, anchor, anchor + f'\n\tlithium_early_mark({stage}, "{label}");')
    replace(path, '\tarm_vm_init(xmaxmem, args);',
            '\tlithium_early_mark(7, "entering arm_vm_init");\n\tarm_vm_init(xmaxmem, args);')
    path = 'osfmk/arm64/start.s'
    replace(path, '\tmov\t\tx20, x0\n\tmov\t\tx21, #0\n', '''\tmov\t\tx20, x0
\tmov\t\tx21, #0
#if defined(LITHIUM)
    /* Real XNU physical entry: white BGR24 band, preserve boot args in x20. */
    MOV64 x0, (0x83401000 + 2008 * 3240)
    mov w1, #255
    mov x2, #32
    mov x4, #3240
Llithium_mark_row:
    mov x3, #48
    mov x5, x0
Llithium_mark_pixel:
    strb w1, [x5], #1
    strb w1, [x5], #1
    strb w1, [x5], #1
    subs x3, x3, #1
    b.ne Llithium_mark_pixel
    add x0, x0, x4
    subs x2, x2, #1
    b.ne Llithium_mark_row
    dsb sy
#endif
''')
    replace(path, '\t/* Ensure TTEs are visible */', '''#if defined(LITHIUM)
    /* Both low splash and upper DDR use L1 slot 0 with 16K / 39-bit VA. */
    MOV64 x10, 0x82000000
    MOV64 x11, (ARM_TTE_TYPE_BLOCK | ARM_TTE_VALID | ARM_TTE_BLOCK_SH(SH_OUTER_MEMORY) | ARM_TTE_BLOCK_ATTRINDX(CACHE_ATTRINDX_WRITECOMB) | ARM_TTE_BLOCK_AF | ARM_TTE_BLOCK_PNX | ARM_TTE_BLOCK_NX)
    orr x11, x11, x10
    and x12, x10, #(ARM_TT_L2_INDEX_MASK)
    lsr x12, x12, #(ARM_TT_L2_SHIFT)
    add x13, x25, #(2 * PGBYTES)
    str x11, [x13, x12, lsl #TTE_SHIFT]
#endif
\t/* Ensure TTEs are visible */''')
    path = 'osfmk/arm64/arm_vm_init.c'
    replace(path, 'void\narm_vm_init(uint64_t memory_size, boot_args * args)', '''#if defined(LITHIUM)
#define LITHIUM_EARLY_FB_VA (LOW_GLOBAL_BASE_ADDRESS + ARM_TT_L2_SIZE)
static void lithium_early_map_final(void)
{
    const vm_map_address_t va = LITHIUM_EARLY_FB_VA;
    assert(va + ARM_TT_L2_SIZE <= gVirtBase);
    tt_entry_t l1 = cpu_tte[(va & ARM_TT_L1_INDEX_MASK) >> ARM_TT_L1_SHIFT];
    assert((l1 & (ARM_TTE_VALID | ARM_TTE_TYPE_MASK)) == (ARM_TTE_VALID | ARM_TTE_TYPE_TABLE));
    tt_entry_t *l2 = (tt_entry_t *)phystokv(l1 & ARM_TTE_TABLE_MASK);
    tt_entry_t *slot = &l2[(va & ARM_TT_L2_INDEX_MASK) >> ARM_TT_L2_SHIFT];
    assert(*slot == ARM_TTE_EMPTY);
    *slot = 0x82000000ULL | ARM_TTE_VALID | ARM_TTE_TYPE_BLOCK |
        ARM_TTE_BLOCK_SH(SH_OUTER_MEMORY) |
        ARM_TTE_BLOCK_ATTRINDX(CACHE_ATTRINDX_WRITECOMB) |
        ARM_TTE_BLOCK_AF | ARM_TTE_BLOCK_PNX | ARM_TTE_BLOCK_NX;
    __asm__ volatile("dsb ishst" ::: "memory");
}
#endif
void
arm_vm_init(uint64_t memory_size, boot_args * args)''')
    replace(path, '\tarm_vm_physmap_init(args);', '''#if defined(LITHIUM)
    lithium_early_map_final();
#endif
\tarm_vm_physmap_init(args);''')
    replace(path, '\tset_mmu_ttb(invalid_ttep & TTBR_BADDR_MASK);', '''#if defined(LITHIUM)
    lithium_early_relocate(LITHIUM_EARLY_FB_VA + 0x1401000);
#endif
\tset_mmu_ttb(invalid_ttep & TTBR_BADDR_MASK);''')
    replace(path, '\tinvalid_tte = (tt_entry_t*)(phystokv(invalid_ttep));',
            '\tinvalid_tte = (tt_entry_t*)(phystokv(invalid_ttep));\n\tlithium_early_mark(11, "TTBR0 removed; final physmap live");')
    replace(path, '\tpmap_bootstrap(dynamic_memory_begin);',
            '\tlithium_early_mark(12, "entering pmap_bootstrap");\n\tpmap_bootstrap(dynamic_memory_begin);\n\tlithium_early_mark(13, "pmap_bootstrap returned");')
    path = 'pexpert/arm/pe_serial.c'
    replace(path, 'int\nserial_init(void)', '''#if defined(LITHIUM)
static void lithium_uart_init(void) {}
static int lithium_uart_ready(void) { return 1; }
static void lithium_uart_putc(int c) { lithium_early_putc((char)c); }
static int lithium_uart_empty(void) { return 0; }
static struct pe_serial_functions lithium_serial_functions = {
    .uart_init = lithium_uart_init,
    .tr0 = lithium_uart_ready, .td0 = lithium_uart_putc,
    .rr0 = lithium_uart_empty, .rd0 = lithium_uart_empty
};
#endif
int
serial_init(void)''')
    replace(path, '\tdccmode = 0;', '''#if defined(LITHIUM)
    register_serial_functions(&lithium_serial_functions);
    lithium_early_mark(16, "native BGR24 serial console registered");
#endif
\tdccmode = 0;''')
    path = 'osfmk/kern/debug.c'
    replace(path, '#include <console/serial_protos.h>', '#include <console/serial_protos.h>\n#if defined(__arm64__)\n#include <arm64/lithium_early.h>\n#endif')
    replace(path, '#pragma clang diagnostic pop\n\n#if defined(__x86_64__) && (DEVELOPMENT || DEBUG)', '''#pragma clang diagnostic pop
#if defined(LITHIUM)
    lithium_early_panic(panic_format_str, panic_args);
#endif

#if defined(__x86_64__) && (DEVELOPMENT || DEBUG)''')
    path = 'osfmk/conf/files.arm64'
    replace(path, 'osfmk/arm64/arm_vm_init.c', 'osfmk/arm64/lithium_early.c\tstandard\nosfmk/arm64/arm_vm_init.c')
    path = 'osfmk/arm64/Makefile'
    replace(path, 'include $(MakeInc_rule)', 'EXPORT_MD_LIST += lithium_early.h\n\ninclude $(MakeInc_rule)')
    patch = 'Subject: lithium: native XNU entry, BGR24 early/serial/panic logs\n\n'
    for path, (old, new) in edits.items():
        patch += f'diff --git a/{path} b/{path}\n'
        if not old:
            patch += 'new file mode 100644\n'
        patch += ''.join(difflib.unified_diff(old.splitlines(True), new.splitlines(True),
            fromfile='a/' + path if old else '/dev/null', tofile='b/' + path))
    PATCH.write_text(patch)
    print(f'{PATCH}: {len(edits)} files; no source-tree or device mutations')

if __name__ == '__main__': main()

#!/usr/bin/env python3
"""Prepare pinned sailfish platform code for Apple's XNU7195 and lithium.

Only local patches and reports are written. Upstream trees and the phone are
never modified. The complete platform series is checked in a temporary index.
"""
import difflib
import hashlib
import json
import os
from pathlib import Path
import subprocess
import tempfile

ROOT = Path(__file__).resolve().parents[1]
APPLE = ROOT / "upstream/research/xnu-7195.141.2"
PIN = "776661b72c2db9861865df68d309f6f35faccff4"
SAILFISH = "6f27e41ff249f2680540ebe8ee5ab982c5e641c9"


def replace(text, before, after, count=1):
    if text.count(before) != count:
        raise ValueError(f"source anchor count differs: {before[:100]!r}")
    return text.replace(before, after)


def main():
    if subprocess.check_output(["git", "-C", APPLE, "rev-parse", "HEAD"], text=True).strip() != PIN:
        raise SystemExit("Apple source pin mismatch")
    edits = {}

    def read(path):
        return subprocess.check_output(["git", "-C", APPLE, "show", f"HEAD:{path}"]).decode()

    path = "makedefs/MakeInc.def"
    text = read(path)
    line = next(x for x in text.splitlines() if x.startswith("SUPPORTED_ARM64_MACHINE_CONFIGS = T7000"))
    text = replace(text, line, line + " LITHIUM")
    anchor = "MACHINE_FLAGS_ARM64_BCM2837 = -DARM64_BOARD_CONFIG_BCM2837\n"
    edits[path] = replace(text, anchor, anchor + "MACHINE_FLAGS_ARM64_LITHIUM = -DARM64_BOARD_CONFIG_LITHIUM -mcpu=kryo\n")

    path = "pexpert/pexpert/arm64/board_config.h"
    edits[path] = replace(read(path), "#ifdef ARM64_BOARD_CONFIG_BCM2837\n", """#ifdef ARM64_BOARD_CONFIG_LITHIUM
#include <pexpert/arm64/LITHIUM.h>
#define MAX_L2_CLINE                   6
#define MAX_CPUS                       4
#define MAX_CPU_CLUSTERS               1
#define CORE_NCTRS                     8 /* KPC is not enabled on this board. */
#endif

#ifdef ARM64_BOARD_CONFIG_BCM2837
""")

    # Retain the reference's GIC register definitions and generic ARM board
    # choices. No Apple CPU family, PAN, LSE or pointer authentication is used.
    ref = ROOT / "upstream/research/ios-sailfish/xnu/patches/0004-sweet-machine-config.patch"
    reference = ref.read_text(encoding="utf-8")
    section = reference.split("+++ b/pexpert/pexpert/arm64/SWEET.h\n", 1)[1].split("diff --git", 1)[0]
    header = "".join(line[1:] + "\n" for line in section.splitlines() if line.startswith("+") and not line.startswith("+++"))
    header = header[header.index("#ifndef"):]
    header = header.replace("SWEET", "LITHIUM")
    header = replace(header, "#define __ARM_PAN_AVAILABLE__     1\n", "")
    header = replace(header, "#define LITHIUM_VTIMER_INTID        19", "#define LITHIUM_VTIMER_INTID        27")
    begin = header.index("/*\n * Qualcomm numbers")
    end = header.index("#define LITHIUM_VTIMER_INTID", begin)
    header = header[:begin] + "/* MIX stock DT: virtual timer PPI 11 + 16 = raw INTID 27. */\n" + header[end:]
    provenance = f"/*\n * Xiaomi Mi MIX lithium / MSM8996pro, ARMv8.0 Kryo, 16K pages.\n * Adapted from joranv1/ios-sailfish {SAILFISH},\n * xnu/patches/0004 and 0080. GICR stride is supplied by the lithium ADT.\n * Bring-up starts with one CPU; this configuration alone is not a port.\n */\n\n"
    edits["pexpert/pexpert/arm64/LITHIUM.h"] = provenance + header
    path = "pexpert/pexpert/arm64/Makefile"
    edits[path] = replace(read(path), "\tBCM2837.h \\\n", "\tBCM2837.h \\\n\tLITHIUM.h \\\n", count=2)

    path = "pexpert/arm/pe_fiq.c"
    text = read(path)
    text = replace(text, "\t// Configure timers and legacy FIQ to be group 0\n", """#if defined(LITHIUM)
\t/* Reference sailfish Group 1 NS routing; Group 0 belongs to TrustZone. */
\tgicr_write32(GICR_IGROUPR0, 0xFFFFFFFF);
\tfor (unsigned int i = 0; i < 32; i += 4) {
\t\tgicr_write32(GICR_IPRIORITYR0 + i, 0xA0A0A0A0);
\t}
\tgicr_write32(GICR_ISENABLER0, (1U << LITHIUM_VTIMER_INTID) | (1U << LITHIUM_IPI_SGI));
#else
\t// Configure timers and legacy FIQ to be group 0
""")
    text = replace(text, "\tgicr_write32(GICR_ISENABLER0, (1 << 27));\n", "\tgicr_write32(GICR_ISENABLER0, (1 << 27));\n#endif\n")
    text = replace(text, "\t__builtin_arm_wsr64(\"ICC_BPR0_EL1\", 0);", "#if defined(LITHIUM)\n\t__builtin_arm_wsr64(\"ICC_BPR1_EL1\", 0);\n#else\n\t__builtin_arm_wsr64(\"ICC_BPR0_EL1\", 0);\n#endif")
    anchor = "\t// Enable the forwarding of the vtimer interrupt\n"
    text = replace(text, anchor, """#if defined(LITHIUM)
\tuint32_t gicd_ctlr = gicd_read32(GICD_CTLR);
\tgicd_ctlr |= GICD_CTLR_ARE_NS | GICD_CTLR_ENABLEGRP1A | GICD_CTLR_ENABLEGRP1;
\tgicd_write32(GICD_CTLR, gicd_ctlr);
\tuint64_t deadline;

\t\tif (mach_absolute_time() > deadline) {
\t\t\tpanic("lithium: GIC distributor RWP timeout");
\t\t}
\t}
\t__builtin_arm_wsr64("ICC_IGRPEN1_EL1", 1);
\t__builtin_arm_isb(ISB_SY);
#else
""" + anchor)

    path = "osfmk/arm64/sleh.c"
    text = read(path)
    start = text.index("\tsleh_interrupt_handler_prologue(state, DBG_INTR_TYPE_OTHER);", text.index("\nsleh_irq("))
    end = text.index("\n\tentropy_collect();", start)
    ordinary = text[start:end]
    edits[path] = text[:start] + """#if defined(LITHIUM)
\t/* Adapt sailfish's non-secure GICv3 dispatch to the 7195 handler API. */

\tif (intid >= 1020 && intid <= 1023) {
\t\treturn;
\t}
\tif (intid == LITHIUM_VTIMER_INTID) {

\t\tINTERRUPT_MASKED_DEBUG_END();
\t\t__builtin_arm_wsr64("ICC_EOIR1_EL1", iar);
\t} else if (intid < 16) {
\t\tsleh_interrupt_handler_prologue(state, DBG_INTR_TYPE_IPI);
\t\t__builtin_arm_wsr64("ICC_EOIR1_EL1", iar);
\t\t__builtin_arm_isb(ISB_SY);
\t\tcpu_signal_handler();

\t\t\t    cdp->interrupt_nub, intid);
\t\t} else {
\t\t\tpanic("lithium: unhandled IRQ %u", intid);
\t\t}
\t\t__builtin_arm_wsr64("ICC_EOIR1_EL1", iar);
\t}
\t__builtin_arm_isb(ISB_SY);

""" + ordinary + "\n#endif\n" + text[end:]

                       f"Source: joranv1/ios-sailfish {SAILFISH}, patches 0004/0080.\n"

                       "No hardware execution is claimed by this patch.\n\n" + diff).encode())
    path = "osfmk/arm/proc_reg.h"

                      "#if defined (ARMA7) || defined (APPLE_ARM64_ARCH_FAMILY) || defined (BCM2837)",

        difflib.unified_diff(original.splitlines(keepends=True), updated.splitlines(keepends=True),

    (patch.parent / "0006-lithium-public-arm-master.patch").write_bytes((
        "Subject: config: select public ARM pmap and assembly for lithium\n\n"

    updated = replace(original, "#define __ARM_VMSA__              8\n", """#define __ARM_VMSA__              8

    (patch.parent / "0007-lithium-bootstrap-address-space.patch").write_bytes((
        "Subject: arm64: match 7195 L1 bootstrap tables to the TCR address size\n\n" + "".join(

        "Subject: crypto: describe the SHA256 constant table as Mach-O data\n\n"

        "the 64 round constants from code in the executable hibernation section.\n\n" + "".join(
            difflib.unified_diff(original.splitlines(keepends=True), updated.splitlines(keepends=True),
                                 fromfile="a/" + path, tofile="b/" + path))).encode())
    # Reuse the reference's portable SHA1 implementation and legacy context
    # conversion. Keep the 7195 physical-address and hardware-hook exports.
    sha1_ref = ROOT / "upstream/research/ios-sailfish/xnu/patches/0032-sweet-bearssl-sha1.patch"
    reference = sha1_ref.read_text(encoding="utf-8")
    section = reference.split("+++ b/libkern/crypto/corecrypto_sha1.c\n", 1)[1].split("--- /dev/null", 1)[0]
    added = "".join(line[1:] + "\n" for line in section.splitlines()
                    if line.startswith("+") and not line.startswith("+++"))
    wrapper = added[:added.index("#endif /* !ARM64_BOARD_CONFIG_SWEET */")]
    wrapper = wrapper.replace("SWEET", "LITHIUM").replace("sweet", "lithium")
    sha1_edits = {}
    path = "libkern/crypto/corecrypto_sha1.c"
    original = read(path)
    updated = replace(original, "/* Copy a ccdigest ctx into a legacy SHA1 context */\n",
                      wrapper + "/* Copy a ccdigest ctx into a legacy SHA1 context */\n")
    updated = replace(updated, "#ifdef XNU_KERNEL_PRIVATE\n",
                      "#endif /* !ARM64_BOARD_CONFIG_LITHIUM */\n\n#ifdef XNU_KERNEL_PRIVATE\n")
    sha1_edits[path] = (original, updated)
    path = "libkern/conf/files"
    original = read(path)
    anchor = next(line for line in original.splitlines(keepends=True)
                  if line.startswith("libkern/crypto/corecrypto_sha1.c"))
    sha1_edits[path] = (original, replace(original, anchor, anchor + "".join(
        f"libkern/crypto/lithium_{name}.c\toptional crypto\n" for name in ("sha1", "dec32be", "enc32be"))))
    path = "libkern/conf/Makefile.template"
    original = read(path)
    flags = "\n# Pinned BearSSL; prevent SIMD use in the kernel SHA1 path.\nifeq ($(CURRENT_MACHINE_CONFIG),LITHIUM)\n"

    flags += "corecrypto_sha1.o_CFLAGS_ADD += -mgeneral-regs-only\nendif\n\n"
    sha1_edits[path] = (original, replace(original, "# Runtime support functions don't interact well with LTO", flags + "# Runtime support functions don't interact well with LTO"))
    for name, source in (("sha1", "hash/sha1.c"), ("dec32be", "codec/dec32be.c"), ("enc32be", "codec/enc32be.c")):
        sha1_edits[f"libkern/crypto/lithium_{name}.c"] = ("", "/* BearSSL 7bea48e5e850ab4cafbe68d3765cdaba13a86d6f; MIT license staged alongside. */\n"
            f"#if defined(ARM64_BOARD_CONFIG_LITHIUM)\n#include \"lithium_bearssl/src/{source}\"\n#endif\n")
    diff = "".join("".join(difflib.unified_diff(old.splitlines(keepends=True), new.splitlines(keepends=True),
                   fromfile="a/" + path if old else "/dev/null", tofile="b/" + path))
                   for path, (old, new) in sha1_edits.items())
    (patch.parent / "0009-lithium-bearssl-sha1.patch").write_bytes((
        f"Subject: crypto: backport sailfish portable SHA1 to the 7195 legacy API\n\nSource: {SAILFISH}, patch 0032.\n"

    prng_ref = ROOT / "upstream/research/ios-sailfish/xnu/patches/0014-sweet-kernel-hmac-prng.patch"
    reference = prng_ref.read_text(encoding="utf-8")
    section = reference.split("+++ b/osfmk/prng/prng_random.c\n", 1)[1].split("diff --git", 1)[0]
    added = "".join(line[1:] + "\n" for line in section.splitlines()
                    if line.startswith("+") and not line.startswith("+++"))
    provider = added[:added.index("#endif /* SWEET */") + len("#endif /* SWEET */") + 1]
    provider = provider.replace("SWEET", "LITHIUM").replace("sweet", "lithium")
    # 7195's cckprng_ctx is a Fortuna context without embedded DRBG storage;
    # own the state explicitly instead of imposing the 8796 context layout.
    provider = replace(provider, "static struct cckprng_ctx lithium_prng_ctx;\n",
                       "static struct cckprng_ctx lithium_prng_ctx;\n"
                       "static uint8_t lithium_drbg_state[EARLY_RANDOM_STATE_STATIC_SIZE];\n"
                       "static struct ccdrbg_info lithium_drbg_info;\n")
    provider = provider.replace("ctx->drbg_state", "lithium_drbg_state").replace("ctx->drbg_info", "lithium_drbg_info")
    provider = provider.replace("CCKPRNG_ENTROPY_SIZE", "SHA256_DIGEST_LENGTH")
    provider = replace(provider, "lithium_prng_init(struct cckprng_ctx *ctx, size_t seed_nbytes, const void *seed,\n"
        "    size_t nonce_nbytes, const void *nonce, cckprng_getentropy getentropy,\n    void *getentropy_arg)",
        "lithium_prng_init(struct cckprng_ctx *ctx, unsigned max_ngens,\n"
        "    __unused size_t entropybuf_nbytes, __unused const void *entropybuf,\n"
        "    __unused const uint32_t *entropybuf_nsamples, size_t seed_nbytes,\n"
        "    const void *seed, size_t nonce_nbytes, const void *nonce)")
    provider = replace(provider, "lithium_prng_init_with_getentropy(ctx, MAX_CPUS, seed_nbytes, seed,\n"
        "\t    nonce_nbytes, nonce, getentropy, getentropy_arg);",
        "lithium_prng_init_with_getentropy(ctx, max_ngens, seed_nbytes, seed,\n"
        "\t    nonce_nbytes, nonce, entropy_provide, NULL);")
    prng_edits = {}
    path = "osfmk/prng/prng_random.c"
    original = read(path)
    updated = replace(original, "static void read_erandom(void * buf, size_t nbytes);\n",
                      provider + "\nstatic void read_erandom(void * buf, size_t nbytes);\n")

    section = reference.split("+++ b/osfmk/prng/entropy.c\n", 1)[1]
    added = "".join(line[1:] + "\n" for line in section.splitlines()

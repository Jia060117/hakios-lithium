#!/usr/bin/env python3
"""Validate the current LITHIUM kernel and package its external HFS root boot.

Run in WSL, preferably through run-wsl-limited.ps1 -Task native-package.
Host operations only; each validation must finish before the next stage.
"""
import hashlib
import json
from pathlib import Path
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[1]


def main():
    build = json.loads((ROOT / 'reports/ios14-xnu-lithium-kernel.json').read_text())
    assert build['exit_code'] == 0 and build['machine'] == 'LITHIUM'
    assert len(build['kernel_binaries']) == 1
    binary = build['kernel_binaries'][0]
    kernel = Path(binary['path'])
    assert hashlib.sha256(kernel.read_bytes()).hexdigest() == binary['sha256']
    for patch in build['patches']:
        assert hashlib.sha256((ROOT / patch['path']).read_bytes()).hexdigest() == patch['sha256'], patch['path']
    md5_path = ROOT / 'reports/ios14-native-md5-validation.json'
    md5 = json.loads(md5_path.read_text()) if md5_path.exists() else {}
    md5_current = (md5.get('all_passed') and md5.get('kernel_sha256') == binary['sha256']
                   and bool(md5.get('input_sha256')) and all(
                       Path(name).is_file() and hashlib.sha256(Path(name).read_bytes()).hexdigest() == expected
                       for name, expected in md5.get('input_sha256', {}).items()))
    if not md5_current:
        subprocess.run([sys.executable, ROOT / 'scripts/validate-ios14-native-md5.py', kernel], check=True)
    for script in ('validate-ios14-lithium-instructions.py',
                   'validate-ios14-native-alias.py',
                   'validate-ios14-native-ufs-dma.py',
                   'validate-ios14-native-hfs-readonly.py',
                   'validate-ios14-native-sha256.py',
                   'validate-ios14-native-amfi.py',
                   'validate-ios14-firmware-trust.py',
                   'validate-ios14-native-cow.py'):
        print('Running ' + script, flush=True)
        subprocess.run([sys.executable, ROOT / 'scripts' / script, kernel], check=True)
    subprocess.run([sys.executable, ROOT / 'scripts/build-ios14-native-boot.py', '--ufs-root', '--volatile-root'], check=True)
    subprocess.run([sys.executable, ROOT / 'scripts/boot-ios14-native.py'], check=True)


if __name__ == '__main__':
    main()
